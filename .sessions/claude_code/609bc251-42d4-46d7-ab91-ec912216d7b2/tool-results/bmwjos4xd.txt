function report(kind, payload = {}) {
  const frame = { kind, ...payload };
  reports.push(frame);
  try {
    if (reportTarget) reportTarget(frame);
    else if (window.parent !== window) window.parent.postMessage({ source: 'lightgame', type: 'lightgame:report', payload: frame }, '*');
  } catch (e) { }
}

function waveComp(w) {
  if (!endless && w <= TOTAL_WAVES) return WAVES[w - 1] || WAVES[0];
  const idx = Math.max(0, w - TOTAL_WAVES - 1);
  const base = WAVES[idx % 10].map(x => [...x]);
  const comp = base.filter(g => g[0] !== 'colossus');
  if (idx % 5 === 4) comp.unshift(['colossus', 1 + Math.floor(idx / 10)]);   // +1 colossus per 2 cycles
  return comp;
}
function wavePlan(w) {
  const idx = Math.max(0, w - TOTAL_WAVES - 1);
  const cycle = Math.floor(idx / 5) + 1;
  // compound +10% per deep wave — maxed towers alone cannot keep up, actives required
  return { hpMult: Math.pow(1.10, idx) * (1 + .16 * cycle), spdMult: 1 + Math.min(.06 * (cycle - 1), .5), cycle, armorAdd: Math.floor(idx / 8) };
}

    endless = false;
    const e = spawn('crawler', .1, { noScale: true, affixes: ['swift'] });
    ok('T24c', e.baseSpeed > ENEMY.crawler.spd && e.affixes.includes('swift') && e.tags.length === 1, `swift spd=${e.baseSpeed.toFixed(2)} tag=${e.tags.length}`);
    G.wipeEnemies();
  }

  // T25 best record persists
  {
    wavesCleared = 7; saveBest();
    ok('T25', best >= 7, `best=${best}`);
  }

  // T26 protocol frame shape
  {
    let got = [];
    G.setReportTarget((f) => got.push(f));
    report('progress', { wave: 3, score: 3 });
    ok('T26', got.length === 1 && got[0].kind === 'progress' && got[0].wave === 3, JSON.stringify(got[0] || {}));
    G.setReportTarget(null);
    reports.length = 0;
  }

  // T27 speed 2× advances sim twice as fast in real frames
  {
    mode = 'build'; callT = 10; G.setSpeed(2);
    G.real(60);
    const after2x = callT;
    G.setSpeed(1);
    G.real(60);
    const after1x = callT;
    ok('T27', Math.abs((10 - after2x) - 2) < .3 && Math.abs((after2x - after1x) - 1) < .3, `2×:${(10 - after2x).toFixed(2)}s 1×:${(after2x - after1x).toFixed(2)}s`);
    callT = 999;
  }

  // T28 boss wave switches score to 'boss'
  {
    endless = true; wave = TOTAL_WAVES + 4;
    G.noEvent(() => callWave(false));
    ok('T28', audio.state === 'boss', `state=${audio.state}`);
    endless = false;
    G.wipeEnemies(); mode = 'build'; callT = 999; audio.setState('build');
  }

  // T29 early call grants +15
  {
    mode = 'build'; callT = 8;
    const before = essence;
    G.noEvent(() => callWave(true));
    ok('T29', essence === before + 15 && mode === 'combat', `+${essence - before}`);
    backToBuild();
  }

  // T30 full 10-wave cruise — no crashes, no NaN, victory reached
  {
    G.reset(); G.noEvent(() => startGame(false));
    let crashed = false;
    try {
      for (let w = 1; w <= TOTAL_WAVES && mode !== 'win'; w++) {
        mode = 'build'; G.noEvent(() => callWave(false));
        R(90);              // let the wave spawn and fight for 1.5s
        G.wipeEnemies();    // clear the remainder (incl. queue)
---tail---
        G.wipeEnemies();    // clear the remainder (incl. queue)
        R(3);               // endWave fires → build / draft / win
        if (mode === 'draft') {
          const c = G.draftOffer();
          if (c.length) G.grant(c[0].id);
          G.ui.hideDraft();
          mode = 'build'; callT = 999;
        }
        if (mode === 'win') break;
      }
    } catch (err) { crashed = true; console.error('T30 crash', err); }
    ok('T30', !crashed && mode === 'win' && !nanFlag, `mode=${mode} nan=${nanFlag} wave=${wave}`);
    G.reset(); G.noEvent(() => startGame(false));
  }

  // T31 build cards open on pad select, close on deselect
  {
    G.select(6);
    const cards = document.getElementById('cards');
    ok('T31', cards.classList.contains('on'), 'cards open');
    deselect();
    ok('T31b', !cards.classList.contains('on'), 'cards closed');
  }

  // T33 flying takes the straight shortcut off the road
  {
    const w = spawn('wisp', 0, { noScale: true });
    ok('T33', w.fly && w.t === 0, 'ground t stays 0');
    for (let i = 0; i < 60; i++) R(1);
    const pathP = world.curve.getPointAt(Math.min(w.flyF, 1));
    const dev = w.group.position.distanceTo(pathP);
    ok('T33b', w.flyF > 0 && dev > 1, `flyF=${w.flyF.toFixed(2)} dev=${dev.toFixed(1)}m`);
    G.wipeEnemies();
  }

  // T34 touch call button fires a wave
  {
    const el = document.getElementById('twave');
    mode = 'build'; callT = 5;
    const before = essence;
    el.dispatchEvent(new PointerEvent('pointerdown', { bubbles: true, pointerType: 'touch' }));
    R(1);
    ok('T34', essence === before + 15 && mode === 'combat', `+${essence - before} mode=${mode}`);
    backToBuild();
  }

  // T35/T36 interaction regressions (with a sane stub viewport — the hidden
  // test guest may report 0x0; the stub also validates the self-heal path)
  {
    const dW = Object.getOwnPropertyDescriptor(window, 'innerWidth');
    const dH = Object.getOwnPropertyDescriptor(window, 'innerHeight');
    Object.defineProperty(window, 'innerWidth', { value: 1280, configurable: true });
    Object.defineProperty(window, 'innerHeight', { value: 720, configurable: true });
    try {
      G.reset(); G.noEvent(() => startGame(false));
      G.step(2);
      world.camera.updateMatrixWorld(true); world.camera.matrixWorldInverse.copy(world.camera.matrixWorld).invert();
      const v = world.pads[3].pos.clone().setY(.5).project(world.camera);
      const px = Math.round((v.x * .5 + .5) * innerWidth), py = Math.round((-v.y * .5 + .5) * innerHeight);
      const cards = () => document.getElementById('cards').classList.contains('on');
      const gl = document.getElementById('gl');
      // touch tap directly on the pad → tap-to-select
      gl.dispatchEvent(new PointerEvent('pointerdown', { bubbles: true, pointerType: 'touch', clientX: px, clientY: py }));
      G.pump();
      ok('T35', cards(), 'touch tap selects pad');
      deselect();
      // stray touch far away, then a mouse click inside the echo window → suppressed
      gl.dispatchEvent(new PointerEvent('pointerdown', { bubbles: true, pointerType: 'touch', clientX: 40, clientY: 40 }));
      G.pump();
      gl.dispatchEvent(new MouseEvent('mousemove', { clientX: px, clientY: py, bubbles: true }));
      gl.dispatchEvent(new MouseEvent('mousedown', { button: 0, clientX: px, clientY: py, bubbles: true }));
      G.pump();
      ok('T35b', !cards(), 'mouse click suppressed inside touch echo window');
      // UI press (target = a DOM button) must not raycast through the DOM and deselect
      G.clickAt(px, py);
      ok('T35c-pre', cards(), 'pad selected via direct click');
      document.getElementById('callbtn').dispatchEvent(new MouseEvent('mousedown', { button: 0, bubbles: true, clientX: px, clientY: py }));
      G.pump();
      ok('T35c', cards(), 'UI press does not deselect');
      deselect();
      // poisoned camera aspect (0x0 collapse) self-heals and picking revives
      const cam = world.camera;
      cam.aspect = NaN; cam.updateProjectionMatrix();
      G.upd(2);
      ok('T36', isFinite(cam.aspect) && Math.abs(cam.aspect - 1280 / 720) < 1e-6, `aspect=${cam.aspect}`);
      const v2 = world.pads[5].pos.clone().setY(.5).project(cam);
      const x = Math.round((v2.x * .5 + .5) * innerWidth), y = Math.round((-v2.y * .5 + .5) * innerHeight);
      G.clickAt(x, y);
      ok('T36b', document.getElementById('cards').classList.contains('on'), 'picking works after heal');
      deselect();
    } finally {
      if (dW) Object.defineProperty(window, 'innerWidth', dW); else delete window.innerWidth;
      if (dH) Object.defineProperty(window, 'innerHeight', dH); else delete window.innerHeight;
    }
  }

  // T37 deep-tide compound scaling
  {
    const p25 = wavePlan(TOTAL_WAVES + 15), p35 = wavePlan(TOTAL_WAVES + 25), p45 = wavePlan(TOTAL_WAVES + 35);
    ok('T37', p25.hpMult > 4 && p35.hpMult > p25.hpMult * 2 && p45.hpMult > p35.hpMult * 2 && p35.armorAdd > p25.armorAdd,
      `25:${p25.hpMult.toFixed(1)} 35:${p35.hpMult.toFixed(1)} 45:${p45.hpMult.toFixed(1)} armor ${p25.armorAdd}→${p35.armorAdd}`);
  }

  // T38 starfall active: cooldown + clustered kill
  {
    G.noEvent(() => callWave(false)); G.wipeEnemies();
    const a = spawn('crawler', .3, { noScale: true }), b = spawn('crawler', .32, { noScale: true }), c = spawn('crawler', .34, { noScale: true });
    const okCast = G.castStarfall();
    ok('T38', okCast && qCd > 0, `cast=${okCast} qCd=${qCd.toFixed(1)}`);
    let dead = false;
    for (let i = 0; i < 60 * 3 && !dead; i++) { R(1); if (!a.alive && !b.alive && !c.alive) dead = true; }
    ok('T38b', dead, `starfall cleared cluster`);
    ok('T38c', !G.castStarfall(), 'on cooldown rejected');
    backToBuild();
  }

  // T39 nova active: freeze then slow
  {
    G.noEvent(() => callWave(false)); G.wipeEnemies();
    const e = spawn('runner', .3, { noScale: true });
    G.castNova();
    ok('T39', e.frozenT > 0 && e.speedNow() === 0, `frozen ${(e.frozenT).toFixed(2)} spd ${e.speedNow()}`);
    const probe = [];
    for (let i = 0; i < 180; i++) {
      step(DT); if (pressed.size) endFrame();
      if (i % 30 === 0) probe.push({ s: +((i + 1) / 60).toFixed(2), fz: +e.frozenT.toFixed(2), al: e.alive, lk: e.leaked, mode, n: enemies.length, hp: +e.hp.toFixed(0), t: +e.t.toFixed(3), q: spawnQ.length });
    }
    ok('T39b', e.frozenT <= 0 && e.slowT > 0 && e.slowF >= .4 && e.speedNow() < e.baseSpeed,
      `after: spd ${e.speedNow().toFixed(2)}/${e.baseSpeed.toFixed(2)} frozen=${e.frozenT.toFixed(2)} slowT=${e.slowT.toFixed(1)} slowF=${e.slowF} alive=${e.alive} probe=${JSON.stringify(probe)}`);
    ok('T39c', !G.castNova(), 'on cooldown rejected');
    backToBuild();
  }

  // T40 overload active: fire-rate surge window
  {
    G.noEvent(() => callWave(false)); G.wipeEnemies();
    G.castOverload();
    const t = towers[0] || G.place(0, 'ballista');
    ok('T40', overchargeT > 8 && t.effRate(overchargeT) < t.effRate(0), `overchargeT=${overchargeT.toFixed(1)} rate ${t.effRate(0).toFixed(2)}→${t.effRate(overchargeT).toFixed(2)}`);
    ok('T40b', !G.castOverload(), 'on cooldown rejected');
    backToBuild();
  }

  // T41 「裂」affix: death spawns two minis, minis do not re-split
  {
    const before = enemies.length;
    const e = spawn('crawler', .3, { noScale: true, hpMult: .3, affixes: ['split'] });
    e.takeHit(9999); if (!e.alive) hooks.onDeath(e);
    R(1);
    const minis = enemies.filter(x => x.mini);
    ok('T41', enemies.length - before === 2 && minis.length === 2 && minis.every(m => m.hp > 0), `+${enemies.length - before} minis`);
    const m0 = minis[0];
    const b2 = enemies.length;
    m0.takeHit(9999); if (!m0.alive) hooks.onDeath(m0);
    R(1);
    ok('T41b', enemies.length === b2 - 1, 'mini does not re-split');
    G.wipeEnemies();
  }

  // T42 「盾」affix: absorb pool then full regen after 4 quiet seconds
  {
    const e = spawn('brute', .5, { noScale: true, affixes: ['shield'] });
    ok('T42', e.shieldMax > 0 && e.shield === e.shieldMax, `shield ${e.shield.toFixed(1)}/${e.shieldMax.toFixed(1)}`);
    const hp0 = e.hp;
    e.takeHit(30);
    ok('T42b', e.shield < e.shieldMax && e.hp < hp0 && e.alive, `hp ${hp0.toFixed(0)}→${e.hp.toFixed(1)} shield ${e.shield.toFixed(1)}`);
    for (let i = 0; i < 60 * 4.5; i++) R(1);
    ok('T42c', e.shield === e.shieldMax, `regen ${e.shield.toFixed(1)}`);
    G.wipeEnemies();
  }

  // T43 「潜」affix: invulnerable dive, towers skip, resurfaces
  {
    const t = towers[0] || G.place(0, 'ballista');
    const e = spawn('crawler', .06, { noScale: true, affixes: ['burrow'] });
    e.burrowT = .1;
    R(8);
    ok('T43', e.burrowed && e.takeHit(50) === 0, `burrowed=${e.burrowed} hit=${e.takeHit(1)}`);
    ok('T43b', t.pick(() => enemies, false) === null || t.pick(() => enemies, false) !== e, 'towers skip burrowed');
    for (let i = 0; i < 60 * 2.2; i++) R(1);
    ok('T43c', !e.burrowed, 'resurfaced');
    G.wipeEnemies();
  }

  // T44 EMP event disables a tower mid-wave
  {
    const t = towers[0] || G.place(0, 'ballista');
    G.noEvent(() => callWave(false)); G.wipeEnemies();
    const filler = spawn('brute', .5, { noScale: true, hpMult: 5 });   // keep the wave alive
    G.forceEvent('emp');
    let hit = false;
    for (let i = 0; i < 60 * 4 && !hit; i++) { R(1); if (towers.some(x => x.disabledT > 0)) hit = true; }
    ok('T44', hit && mode === 'combat', `EMP blackout (mode=${mode})`);
    G.revertEvent('emp');
    towers.forEach(x => { x.disabledT = 0; x.setDisabled(false); });
    backToBuild();
  }

  // T45 colossus count escalates per cycle
  {
    endless = true;
    ok('T45', waveComp(TOTAL_WAVES + 15).filter(g => g[0] === 'colossus')[0][1] === 2 && waveComp(TOTAL_WAVES + 25).filter(g => g[0] === 'colossus')[0][1] === 3, '2 → 3 colossi');
    endless = false;
  }

  // T46 corrode/frenzy wave mods reach spawns
  {
    endless = true; wave = TOTAL_WAVES + 5;
    G.forceEvent('corrode');
    const e1 = spawn('crawler', .1, { noScale: true, affixes: [] });
    ok('T46', e1.armor === ENEMY.crawler.armor + 3, `armor ${e1.armor}`);
    G.revertEvent('corrode');
    G.forceEvent('frenzy');
    const e2 = spawn('crawler', .1, { noScale: true, affixes: [] });
    ok('T46b', e2.baseSpeed > ENEMY.crawler.spd * 1.25, `spd ${e2.baseSpeed.toFixed(2)}`);
    G.revertEvent('frenzy');
    endless = false;
    G.wipeEnemies();
  }

  // wrap: back to title
  G.reset();
  await nap();
  ok('DONE', T.passed + T.failed >= 30, `passed=${T.passed} failed=${T.failed}`);
  console.log(`__TEST ${T.passed} passed, ${T.failed} failed`);
}
